class Solution {
    public String mapWordWeights(String[] words, int[] weights) {
       // here we creating stringBuilder to store resultent array
        StringBuilder sb= new StringBuilder();
        for(String i : words){
            int sum=0;
            char ch[]= i.toCharArray();
            for(char c : ch){
                sum+=weights[c-'a'];
            }
            int l=sum%26;
            char res=(char)('z'-l);
            sb.append(res);
        }
        return sb.toString();
    }
